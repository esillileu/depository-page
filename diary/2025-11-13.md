		```bash
		cmake -B build \
		  -DGGML_BLAS=ON \
		  -DGGML_BLAS_VENDOR=Intel10_64lp \
		  -DCMAKE_C_COMPILER=icx \
		  -DCMAKE_CXX_COMPILER=icpx \
		  -DGGML_NATIVE=ON \
		  -DBUILD_SHARED_LIBS=OFF
		cmake --build build --config Release
		```
